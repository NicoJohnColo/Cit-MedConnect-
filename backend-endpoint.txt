Add this endpoint to your UserController.java:

@PutMapping("/school-id/{schoolId}")
public ResponseEntity<?> updateUserBySchoolId(@PathVariable String schoolId, @RequestBody UserDTO userDTO) {
    try {
        UserDTO updatedUser = userService.updateUserBySchoolId(schoolId, userDTO);
        return ResponseEntity.ok(updatedUser);
    } catch (RuntimeException e) {
        return ResponseEntity.status(HttpStatus.NOT_FOUND)
                .body(Map.of("error", e.getMessage()));
    }
}

And add this method to your UserService.java:

public UserDTO updateUserBySchoolId(String schoolId, UserDTO userDTO) {
    User user = userRepository.findBySchoolId(schoolId)
        .orElseThrow(() -> new RuntimeException("User not found with schoolId: " + schoolId));
    
    // Update user fields
    user.setFirstName(userDTO.getFirstName());
    user.setLastName(userDTO.getLastName());
    user.setPhone(userDTO.getPhone());
    user.setAge(userDTO.getAge());
    user.setGender(userDTO.getGender());
    user.setEmail(userDTO.getEmail());
    user.setRole(userDTO.getRole());
    
    User updatedUser = userRepository.save(user);
    return convertToDTO(updatedUser);
}
